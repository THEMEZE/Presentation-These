  \frametitle{Velocities vs Rapidities }
  {\fontsize{10pt}{5}\selectfont
  The rapidities are the \textbf{asymptotic velocities, when one lets the system expand in 1D :}\\[0.1cm]
  \begin{center}
    \includegraphics[width=0.4\textwidth, page=11]{figures/Figures.pdf}
  \end{center}

  \begin{center}
    $$ \tilde{n}(v,\tau ) \underset{\tau \to \infty}{\rightarrow} L\rho(v) $$
  \end{center}

  \onslide<2->{
  \textbf{The rapidity distribution is an experimentally accessible quantity}, homothetic to the asymptotic density profile following a longitudinal expansion.

  \begin{center}
    $$ \tau n(x, \tau )  \underset{\tau \to \infty}{\rightarrow}  \tilde{n}(x/\tau) \underset{\tau \to \infty}{\rightarrow} L \rho(x/\tau) $$
  \end{center}
  }
  
  }

\begin{frame}
  \frametitle{Generalized Gibbs Ensemble (GGE) for a Homogeneous, Stationary Subsystem}

  {\fontsize{8}{7}\selectfont
  %\textbf{Homogeneous system:} Lieb–Liniger model \\[4pt]
  \textbf{After relaxation:} described by a \textit{Generalized Gibbs Ensemble (GGE)} \\[6pt]
  $$ \operator{\varrho}_{\mathrm{GGE}} \propto e^{-\operator{Q}} , \quad  \langle\{\theta_a\}\vert e^{\displaystyle -\operator{Q}}\vert \{\theta_a\}\rangle = e^{-\sum_{a = 1}^N \beta(\theta)}$$


  \textbf{Key idea:}  
  \begin{itemize}
    \item Integrable systems $\Rightarrow$ infinitely many conserved quantities.
    \item GGE $\Rightarrow$ statistical ensemble accounting for all conserved charges.
  \end{itemize}

  \vspace{4pt}
  \textbf{Open question:}

    \item Several ensembles can yield the same $ \overline{\rho}(\theta)$.  
    \item Need to study \textbf{fluctuations} around $\overline{\rho}$:  
      $$\rho = \overline{\rho} + \delta\rho$$
  \end{itemize}
